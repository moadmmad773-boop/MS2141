'PM' : 'AM'}`; } setInterval(updateClock, 1000); updateClock(); socket.emit('registerScreen'); socket.on('screenRegistered', (data) => { document.getElementById('pin-display').innerText = data.pin; const remoteUrl = `${window.location.origin}/remote.html?pin=${data.pin}`; document.getElementById('qrcode').innerHTML = ''; new QRCode(document.getElementById('qrcode'), { text: remoteUrl, width: 85, height: 85 }); const banner = document.getElementById('pair-banner'); banner.classList.add('show'); setTimeout(() => banner.classList.remove('show'), 5000); }); // تهيئة مشغل يوتيوب function onYouTubeIframeAPIReady() { player = new YT.Player('yt-player', { height: '100%', width: '100%', playerVars: { 'autoplay': 1, 'controls': 1, 'playsinline': 1, 'enablejsapi': 1, 'rel': 0 }, events: { 'onReady': onPlayerReady, 'onStateChange': onPlayerStateChange } }); } function onPlayerReady(event) { playerReady = true; if (pendingVideoId) { startVideo(pendingVideoId); pendingVideoId = null; } } // معالجة حالة التوقف القسري وإجبار التشغيل function onPlayerStateChange(event) { if (event.data === YT.PlayerState.PAUSED && !audioUnlocked) { // إذا قام المتصفح بإيقافه قسرياً، شغل الفيديو مكتوماً فوراً لضمان استمرار العرض player.mute(); player.playVideo(); } } function openYoutube() { document.getElementById('youtube-screen').classList.add('active'); } function closeYoutube() { document.getElementById('youtube-screen').classList.remove('active'); if (player && player.stopVideo) player.stopVideo(); } function startVideo(videoId) { openYoutube(); if (!playerReady) { pendingVideoId = videoId; return; } // إذا لم يفك الجوال حظر الصوت بعد، نشغل بشكل مكتوم لمنع التوقف عند الثانية 1 if (!audioUnlocked) { player.loadVideoById({ videoId: videoId }); player.mute(); player.playVideo(); } else { player.loadVideoById({ videoId: videoId }); player.unMute(); player.setVolume(100); player.playVideo(); } } // استقبال أمر التشغيل من الريموت socket.on('playYoutube', (data) => { const vId = typeof data === 'string' ? data : data.videoId; if (vId) startVideo(vId); }); socket.on('stateUpdate', (state) => { if (state.currentApp === 'home') closeYoutube(); if (state.currentApp === 'youtube' && state.youtubeId) { startVideo(state.youtubeId); } }); // بمجرد التفاعل مع الريموت يتم فك الحظر عن الصوت نهائياً لشاشة السيارة socket.on('remoteNavigate', (dir) => { audioUnlocked = true; // تم اعتماد إذن التفاعل if (player && playerReady) { player.unMute(); player.setVolume(100); if (dir === 'select') { const state = player.getPlayerState(); if (state === YT.PlayerState.PLAYING) player.pauseVideo(); else player.playVideo(); } } });
